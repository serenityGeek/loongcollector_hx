docker pull sls-opensource-registry.cn-shanghai.cr.aliyuncs.com/loongcollector-community-edition/loongcollector-build-linux:2.1.15


##1 运行容器
docker run --name loongcollector-build-2.1.15 -d -v /home/shsnc/zhang/snc-agent-loogcollector-plugin/loongcollector:/loongcollector -w /loongcollector sls-opensource-registry.cn-shanghai.cr.aliyuncs.com/loongcollector-community-edition/loongcollector-build-linux:2.1.15 bash -c "sleep infinity"

##2 进入容器
docker exec -it -u root loongcollector-build-2.1.15 bash

##3 编译C++代码
git submodule update --init
mkdir -p core/build
cd core/build
cmake -DJAVA_HOME=/opt/jdk1.8.0_202 ..
make -sj$(nproc)

##4 编译go代码
go mod tidy       # 若需要更新插件库
make plugin_local # 每次更新插件代码后从这里开始

##5 运行loongCollector
export LD_LIBRARY_PATH=/home/shsnc/zhang/loongcollector-custom/:$LD_LIBRARY_PATH
export LD_PRELOAD=/home/shsnc/zhang/loongcollector-custom/libloongcollector.so
java -Djava.library.path=/home/shsnc/zhang/loongcollector-custom -cp /home/shsnc/zhang/loongcollector-custom/bin com.shsnc.agent.ivory.plugin.loongcollector.LoongCollectorProcessor

##6 编译产出
cp -a ./core/build/libloongcollector.so ./output ; cp -a ./core/build/go_pipeline/libGoPluginAdapter.so ./output ; cp -a core/build/ebpf/driver/libeBPFDriver.so ./output ; cp  core/build/_thirdparty/coolbpf/src/libcoolbpf.so ./output
